cmake_minimum_required(VERSION 3.16)
project(SubtitleCore LANGUAGES CXX)
add_library(subtitle_core STATIC src/core.cpp src/text.cpp)
target_include_directories(subtitle_core PUBLIC include)
target_compile_features(subtitle_core PUBLIC cxx_std_17)
set_target_properties(subtitle_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(subtitle_core PRIVATE /W4 /utf-8)
endif()
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  include(CTest)
endif()
if(BUILD_TESTING AND NOT ANDROID AND NOT IOS)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE subtitle_core)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8)
  endif()
  add_test(NAME shared_core COMMAND core_tests)
endif()
