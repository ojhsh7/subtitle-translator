cmake_minimum_required(VERSION 3.21)
project(SubtitleTranslator VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_subdirectory(core)
if(NOT WIN32)
  message(STATUS "Desktop UI is Windows-only. Building the portable C++ core.")
  return()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test Concurrent)
qt_standard_project_setup()
# ONNX Runtime runs the PaddleOCR recognition models (official onnxruntime-win-x64 package, MIT).
set(ONNXRUNTIME_ROOT "" CACHE PATH "Folder of the onnxruntime-win-x64 package (include/, lib/)")
if(NOT EXISTS "${ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the extracted onnxruntime-win-x64 package.")
endif()
add_library(subtitle_services STATIC
  src/ocr_service.cpp src/ocr_service.h
  src/translation_service.cpp src/translation_service.h
  src/credential_store.cpp src/credential_store.h
  src/capture_service.cpp src/capture_service.h
  src/change_detector.cpp src/change_detector.h
  src/languages.cpp src/languages.h
  src/paddle_ocr.cpp src/paddle_ocr.h
  src/game_window.cpp src/game_window.h)
target_include_directories(subtitle_services PUBLIC src)
target_include_directories(subtitle_services PRIVATE "${ONNXRUNTIME_ROOT}/include")
target_link_libraries(subtitle_services PUBLIC Qt6::Widgets Qt6::Network subtitle_core PRIVATE Advapi32 Psapi User32 Dwmapi icu Qt6::Concurrent "${ONNXRUNTIME_ROOT}/lib/onnxruntime.lib")
qt_add_executable(SubtitleTranslator WIN32
  src/main.cpp src/main_window.cpp src/main_window.h
  src/region_selector.cpp src/region_selector.h src/subtitle_overlay.cpp src/subtitle_overlay.h)
target_link_libraries(SubtitleTranslator PRIVATE subtitle_services subtitle_core Qt6::Concurrent User32)
# Windows ships an older onnxruntime.dll in System32; ours must sit next to every executable.
add_custom_target(onnxruntime_runtime ALL
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll"
          "${ONNXRUNTIME_ROOT}/lib/onnxruntime_providers_shared.dll" "$<TARGET_FILE_DIR:SubtitleTranslator>")
add_dependencies(onnxruntime_runtime SubtitleTranslator)
if(MSVC)
  target_compile_options(subtitle_services PRIVATE /W4 /utf-8)
  target_compile_options(SubtitleTranslator PRIVATE /W4 /utf-8)
endif()
include(CTest)

# Version information in every executable's file properties.
function(add_version_resource target description)
  set(RC_NAME ${target})
  set(RC_DESCRIPTION "${description}")
  configure_file(${CMAKE_CURRENT_SOURCE_DIR}/src/version.rc.in ${CMAKE_CURRENT_BINARY_DIR}/${target}_version.rc @ONLY)
  target_sources(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${target}_version.rc)
endfunction()
add_version_resource(SubtitleTranslator "Subtitle Translator - game subtitle OCR and translation")
