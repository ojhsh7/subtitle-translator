cmake_minimum_required(VERSION 3.21)
project(SubtitleTranslator VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_subdirectory(core)
if(NOT WIN32)
  message(STATUS "Desktop UI is Windows-only. Building the portable C++ core.")
  return()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test Concurrent)
qt_standard_project_setup()
add_library(subtitle_services STATIC
  src/ocr_service.cpp src/ocr_service.h
  src/translation_service.cpp src/translation_service.h
  src/credential_store.cpp src/credential_store.h
  src/capture_service.cpp src/capture_service.h
  src/change_detector.cpp src/change_detector.h
  src/languages.cpp src/languages.h)
target_include_directories(subtitle_services PUBLIC src)
target_link_libraries(subtitle_services PUBLIC Qt6::Widgets Qt6::Network PRIVATE Advapi32 Psapi Qt6::Concurrent)
qt_add_executable(SubtitleTranslator WIN32
  src/main.cpp src/main_window.cpp src/main_window.h
  src/region_selector.cpp src/region_selector.h src/subtitle_overlay.cpp src/subtitle_overlay.h)
target_link_libraries(SubtitleTranslator PRIVATE subtitle_services subtitle_core Qt6::Concurrent User32)
if(MSVC)
  target_compile_options(subtitle_services PRIVATE /W4 /utf-8)
  target_compile_options(SubtitleTranslator PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(translation_tests tests/translation_tests.cpp)
  target_link_libraries(translation_tests PRIVATE subtitle_services Qt6::Test)
  if(MSVC)
    target_compile_options(translation_tests PRIVATE /utf-8)
  endif()
  add_test(NAME translation_protocol COMMAND translation_tests)
  qt_add_executable(integration_checks tests/integration_checks.cpp)
  target_link_libraries(integration_checks PRIVATE subtitle_services Qt6::Test)
  if(MSVC)
    target_compile_options(integration_checks PRIVATE /utf-8)
  endif()
  add_test(NAME local_checks COMMAND integration_checks)
  qt_add_executable(change_detector_tests tests/change_detector_tests.cpp)
  target_link_libraries(change_detector_tests PRIVATE subtitle_services Qt6::Test)
  if(MSVC)
    target_compile_options(change_detector_tests PRIVATE /utf-8)
  endif()
  add_test(NAME change_detection COMMAND change_detector_tests)
endif()
install(TARGETS SubtitleTranslator RUNTIME DESTINATION .)


qt_add_executable(subtitle_cli src/pipeline_cli.cpp)
target_link_libraries(subtitle_cli PRIVATE subtitle_services subtitle_core Psapi)
if(MSVC)
  target_compile_options(subtitle_cli PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
  qt_add_executable(window_checks tests/window_checks.cpp src/main_window.cpp src/main_window.h src/region_selector.cpp src/region_selector.h src/subtitle_overlay.cpp src/subtitle_overlay.h)
  target_link_libraries(window_checks PRIVATE subtitle_services subtitle_core Qt6::Concurrent Qt6::Test User32)
  if(MSVC)
    target_compile_options(window_checks PRIVATE /utf-8)
  endif()
  add_test(NAME windows_flow COMMAND window_checks -o "${CMAKE_CURRENT_BINARY_DIR}/windows-flow.txt,txt")
endif()
